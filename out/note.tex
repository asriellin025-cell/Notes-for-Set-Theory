    $B\subseteq C \implies \mathscr{P}(B) \subseteq \mathscr{P}(C) $
    \begin{proof}

        By def $\mathscr{P}(B) = \left\{ x \mid x \subseteq B \right\}$

        $\forall x, x \in \mathscr{P}(B)$, Then, $x \subseteq B \subseteq C$

        $\implies x \subseteq C$
    
        $\implies x \in \mathscr{P}(C)$ By def $\mathscr{P} (C) = \left\{ x \mid x \subseteq C \right\}$

        Thus, $\mathscr{P}(B) \subseteq \mathscr{P}(C)$

    \end{proof}

    A is a finite set,$|A|=n \implies |\mathscr{P}(A)| = 2^n$
    \begin{proof}
        
        Consider taking $i$ elements from A to build a subset.

        Then $i \in \left\{ 0, \dots, n\right\}$, and the number of that kind of subsets is $\binom{n}{i}$.

        (if $i = 0$, the subset is $\emptyset$; if $i = n$, the subset is A itself)

        $\implies |\mathscr{P}(A)| = \sum_{i = 0}^{n}\binom{n}{i}$

        $\implies |\mathscr{P}(A)| = 2^n$, By Binomial Theorem, consider $(1+1)^n$

    \end{proof}

    $x, y \in B \implies \left\{ \left\{ x\right\},\left\{x, y\right\} \right\} 
    \in \mathscr{P}(\mathscr{P}(B))$
    \begin{proof}
        $x, y \in B$

        $\implies \left\{x\right\}, \left\{ x, y\right\} \in \mathscr{P}(B)$

        $\implies \left\{\left\{x\right\}, \left\{x, y\right\}\right\} \subseteq \mathscr{P}(B)$

        Thus, by def $\left\{\left\{x\right\}, \left\{x, y\right\}\right\} \in \mathscr{P}(\mathscr{P}(B))$
    \end{proof}

    \begin{remark}
        A simple way to calculate the rank of set (which finite) is to count the deepest nested brackets until meet atom or $\emptyset$.
    \end{remark}

    $\forall \alpha \in \mathbb{N},\alpha \le 3, V_{\alpha + 1} = A \cup \mathscr{P}(V_{\alpha})$
    \begin{proof}
        $V_0 = A$

        By def, $V_{n+1} = V_n \cup \mathscr{P}(V_n)$
        
        $V_1 = V_0 \cup \mathscr{P}(V_0) = A \cup \mathscr{P}(V_0)$

        $V_2 = V_1 \cup \mathscr{P}(V_1) = A \cup \mathscr{P}(V_0) \cup  \mathscr{P}(V_1) = A \cup \mathscr{P}(V_1)$
        
        $\mathscr{P}(V_0) \cup  \mathscr{P}(V_1) =  \mathscr{P}(V_1)$,
        By $\mathscr{P}(V_0) \subseteq \mathscr{P}(V_1)$ since $V_0 \subseteq V_1$, by previous result.

        $V_3 = V_2 \cup \mathscr{P}(V_2) = A \cup \mathscr{P}(V_1) \cup  \mathscr{P}(V_2)$ Similarly,

        $V_3 = A \cup \mathscr{P}(V_2)$
    \end{proof}

    \begin{remark}
        we can use mathematical induction to make it to $\forall \alpha \in \mathbb{N}$
    \end{remark}

    \begin{axiom}
        Extensionality Axiom
        $$\forall A \forall B ,(\forall x, x \in A \iff x \in B) \implies A = B$$
    \end{axiom}

    \begin{axiom}
        Empty Set Axiom
        $$\exists A \forall x, x \notin A$$
        \begin{remark}
            Does $\emptyset$ act like a unique generator in terms of abstract algebra?

             (e.g. similar as prime numbers can be used to generate whole $\mathbb{N}$ under multiplication)
        \end{remark}
    \end{axiom}

    \begin{axiom}
        Pairing Axiom
        $$\forall A \forall B \exists C \forall x(x \in C \iff x=A \vee x=B)$$ 
    \end{axiom}

    \begin{axiom}
        Union Axiom
        $$\forall A \exists B \forall x(x \in B \iff (\exists b \in A)x\in b)$$
        \begin{remark}
            Compare with $\forall A \forall B \exists C \forall x(x\in C \iff x\in A \vee x\in B)$,

            Which mentioned in previous page of the book (Page 18), named Preliminary Form. 
            Preliminary Form build union as a binary operator, similar to the Pairing Axiom.
            While the problem of Preliminary Form is can not tackle infinity sets 
            (you have to nest the union operation infinity many times to do that, which is illegal.
             A Well-Formed Formula in First-Order Classical Logic should have finite length)
        \end{remark}
    \end{axiom}

    \begin{axiom}
        Power Set Axiom
        $$\forall A \exists B \forall x(x\in B \iff x \subseteq A)$$
    \end{axiom}

    \begin{axiom}
        Subset Axiom
        $$\forall t_1 \dots \forall t_n \forall c \exists B \forall x(x\in B \iff x \in c \wedge \phi(x,t_1,\dots,t_n))$$
        \begin{remark}
            It is an axiom schema, where $\phi$ is a first-order formula;
            
            The number of subset axioms would be $\aleph_0$;

            $P(x)$ (predicate) includes $\phi(x)$ (open formula) and $\sigma$ (closed formula).
        \end{remark}
    \end{axiom}
